% Feature map block with 3D effect
% #1: x, #2: y, #3: height, #4: width (depth), #5: color, #6: label
\newcommand{\featuremap}[6]{
    \begin{scope}[shift={(#1, #2)}]
        % Back face
        \fill[#5!50!black] (0.1, 0.1) rectangle ++(#4, #3);
        % Front face
        \fill[#5] (0, 0) rectangle ++(#4, #3);
        % Top edge
        \fill[#5!80!white] (0, #3) -- ++(0.1, 0.1) -- ++(#4, 0) -- ++(-0.1, -0.1) -- cycle;
        % Right edge
        \fill[#5!60!black] (#4, 0) -- ++(0.1, 0.1) -- ++(0, #3) -- ++(-0.1, -0.1) -- cycle;
        % Border
        \draw[black, thick] (0, 0) rectangle ++(#4, #3);
        \draw[black] (0, #3) -- ++(0.1, 0.1) -- ++(#4, 0);
        \draw[black] (#4, 0) -- ++(0.1, 0.1) -- ++(0, #3);
        % Label
        \node[font=\tiny, below, align=center] at (#4/2, -0.1) {#6};
    \end{scope}
}

% Stacked feature maps
\newcommand{\stackedmaps}[7]{
    % #1: x, #2: y, #3: height, #4: width, #5: color, #6: num stacks, #7: label
    \begin{scope}[shift={(#1, #2)}]
        \foreach \i in {1,...,#6} {
            \pgfmathsetmacro{\offset}{(\i-1)*0.06}
            \fill[#5!50!black, opacity=0.9] (\offset+0.1, \offset+0.1) rectangle ++({#4}, {#3});
            \fill[#5, opacity=0.95] (\offset, \offset) rectangle ++({#4}, {#3});
            \fill[#5!80!white, opacity=0.9] (\offset, {#3+\offset}) -- ++(0.1, 0.1) -- ++({#4}, 0) -- ++(-0.1, -0.1) -- cycle;
            \fill[#5!60!black, opacity=0.9] ({#4+\offset}, \offset) -- ++(0.1, 0.1) -- ++(0, {#3}) -- ++(-0.1, -0.1) -- cycle;
        }
        \pgfmathsetmacro{\finaloff}{(#6-1)*0.06}
        \draw[black, thick] (\finaloff, \finaloff) rectangle ++({#4}, {#3});
        \draw[black] (\finaloff, {#3+\finaloff}) -- ++(0.1, 0.1) -- ++({#4}, 0);
        \draw[black] ({#4+\finaloff}, \finaloff) -- ++(0.1, 0.1) -- ++(0, {#3});
        \node[font=\tiny, below, align=center] at ({#4/2 + \finaloff/2}, -0.15) {#7};
    \end{scope}
}

% Operation box
\newcommand{\opbox}[5]{
    % #1: x, #2: y, #3: width, #4: color, #5: label
    \fill[#4, rounded corners=2pt] (#1, #2) rectangle ++(#3, 0.4);
    \draw[black, rounded corners=2pt] (#1, #2) rectangle ++(#3, 0.4);
    \node[font=\tiny\bfseries, white] at ({#1 + #3/2}, {#2 + 0.2}) {#5};
}

% FC layer (vertical bar)
\newcommand{\fclayer}[5]{
    % #1: x, #2: y, #3: height, #4: color, #5: label
    \fill[#4] (#1, #2) rectangle ++(0.15, #3);
    \draw[black, thick] (#1, #2) rectangle ++(0.15, #3);
    \node[font=\tiny, below, align=center] at ({#1 + 0.075}, {#2 - 0.1}) {#5};
}
